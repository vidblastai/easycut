<script>
// A soft, slow push like turning a page; the lines boil a little.
tl.fromTo("#world", { scale: 1.3 }, { scale: 1.38, transformOrigin: "540px 620px", duration: DURATION, ease: "none" }, 0);
tl.to("#boil", { x: 1.5, y: -1, duration: .1, yoyo: true, repeat: 27, ease: "steps(1)" }, 0);

// Hook: the night sky blooms and the lighthouse rises into it — "every night".
tl.fromTo("#sky", { scale: .3, opacity: 0, svgOrigin: "540 620" }, { scale: 1, opacity: 1, svgOrigin: "540 620", duration: .55, ease: "back.out(1.3)" }, 0);
tl.fromTo("#tower", { y: 500 }, { y: 0, duration: .5, ease: "back.out(1.4)" }, .1);
tl.fromTo("#sea", { y: 160, opacity: 0 }, { y: 0, opacity: 1, duration: .4, ease: "power2.out" }, .15);
tl.to(".star", { opacity: .3, duration: .2, yoyo: true, repeat: 7, stagger: .1, ease: "steps(1)" }, .5);
tl.to(".wave", { x: 20, duration: .6, yoyo: true, repeat: 7, ease: "sine.inOut" }, 0);

// "the old lighthouse keeper": his lantern appears at the door.
const keeper = at("keeper", 1.5) - .2;
tl.fromTo("#keeper", { x: 540, y: 940, scale: 0 }, { x: 540, y: 940, scale: 1, duration: .3, ease: "back.out(2)" }, keeper);

// "climbs ninety nine steps": up the zigzag, a step ticking in under each footfall.
const climb = at("climbs", 1.9) - .05;
const pts = [[500, 925], [570, 890], [510, 847], [570, 810], [510, 767], [570, 727], [515, 685], [569, 645], [517, 605], [565, 565], [519, 525], [561, 485], [540, 420]];
pts.forEach((p, i) => {
  const t = climb + i * .13;
  tl.to("#keeper", { x: p[0], y: p[1], duration: .12, ease: "power1.inOut" }, t);
  if (i < 12) {
    tl.set(".stp:nth-child(" + (i + 1) + ")", { opacity: 1 }, t + .06);
  }
});

// "switch on the light": the lamp room fills with yellow and the beams swing out.
const sw = at("switch", 3.6) - .1;
tl.to("#keeper", { scale: 0, duration: .15, ease: "power2.in" }, sw - .05);
tl.to("#glowLamp", { opacity: 1, duration: .08, yoyo: true, repeat: 2, ease: "steps(1)" }, sw);
tl.set("#glowLamp", { opacity: 1 }, sw + .3);
const light = at("light", 4.14) - .15;
tl.set("#beamL, #beamR", { opacity: .9 }, light);
tl.fromTo("#beamL", { scaleX: 0, svgOrigin: "540 372" }, { scaleX: 1, svgOrigin: "540 372", duration: .4, ease: "power3.out", immediateRender: false }, light);
tl.fromTo("#beamR", { scaleX: 0, svgOrigin: "540 372" }, { scaleX: 1, svgOrigin: "540 372", duration: .4, ease: "power3.out", immediateRender: false }, light + .05);
tl.to("#sky", { opacity: .8, duration: .4 }, light);

// "for the ships": the left beam dips onto a ship sailing in, which bobs in its light.
const ships = at("ships", 4.64) - .2;
tl.set("#ship", { opacity: 1 }, ships);
tl.fromTo("#ship", { x: -260 }, { x: 0, duration: .5, ease: "power3.out", immediateRender: false }, ships);
tl.to("#beamL", { rotation: -14, svgOrigin: "540 372", duration: .5, ease: "power2.inOut" }, ships);
tl.to("#ship", { y: -8, rotation: -2, svgOrigin: "240 1000", duration: .25, yoyo: true, repeat: 3, ease: "sine.inOut" }, ships + .5);
</script>
