<script>
// One smooth, continuous push; nothing else moves the camera.
tl.fromTo("#world", { scale: 1.08 }, { scale: 1.16, transformOrigin: "540px 630px", duration: DURATION, ease: "none" }, 0);

// Hook: the noise wipes in from the left, already crowded — "cut the noise".
tl.to("#noiseInRect", { attr: { width: 840 }, duration: .35, ease: "expo.out" }, 0);
tl.to(".nz", { x: (i) => (rand(i) - .5) * 30, y: (i) => (rand(i + 50) - .5) * 30, rotation: (i) => (rand(i + 90) - .5) * 30, transformOrigin: "50% 50%", duration: .6, ease: "none" }, 0);

// The lime blade drops through the frame and the noise below it is gone.
const cut = at("noise", .62);
tl.set("#blade", { opacity: 1 }, cut);
tl.to("#blade", { attr: { y: 980 }, duration: .45, ease: "expo.inOut" }, cut);
tl.to("#noiseCutRect", { attr: { y: 960, height: 0 }, duration: .45, ease: "expo.inOut" }, cut);
tl.to("#blade", { attr: { x: 540, width: 0 }, duration: .25, ease: "expo.in" }, cut + .45);

// "keep one idea": a single circle is unmasked from its centre line, exactly where the cut ended.
const idea = Math.min(at("one", 1.4) - .15, cut + .45);
tl.set("#idea", { opacity: 1 }, idea);
tl.to("#ideaInRect", { attr: { y: 380, height: 500 }, duration: .55, ease: "expo.out" }, idea);
tl.fromTo("#core", { scale: 0, svgOrigin: "540 630" }, { scale: 1, duration: .4, ease: "expo.out", immediateRender: false }, at("idea", 1.62));

// "give it room": the corner brackets snap on tight, then open wide.
const room = at("room", 2.62) - .3;
tl.set("#brackets", { opacity: 1 }, room);
tl.fromTo("#brackets", { scale: 1.3, svgOrigin: "540 630" }, { scale: 1, duration: .25, ease: "expo.out", immediateRender: false }, room);
tl.to("#bTL", { x: -150, y: -200, duration: .7, ease: "expo.inOut" }, room + .3);
tl.to("#bTR", { x: 150, y: -200, duration: .7, ease: "expo.inOut" }, room + .3);
tl.to("#bBL", { x: -150, y: 200, duration: .7, ease: "expo.inOut" }, room + .3);
tl.to("#bBR", { x: 150, y: 200, duration: .7, ease: "expo.inOut" }, room + .3);

// "to breathe": the idea takes one slow breath in, and holds.
const breathe = at("breathe", 2.94);
tl.to("#idea", { scale: 1.18, svgOrigin: "540 630", duration: DURATION - breathe, ease: "power2.out" }, breathe);
tl.to("#core", { scale: 1.6, svgOrigin: "540 630", duration: .5, ease: "expo.out" }, breathe);
</script>
