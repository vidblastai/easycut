<script>
// The screen hums: the picture breathes a touch and the beam flickers.
tl.fromTo("#world", { scale: 1.02 }, { scale: 1.08, transformOrigin: "540px 640px", duration: DURATION, ease: "none" }, 0);
tl.to(".beam", { opacity: .86, duration: .05, yoyo: true, repeat: 9, ease: "none" }, 1.2);

// Hook: the beam starts tracing the heart at once and closes it on "heart".
const heart = at("heart", .45);
tl.to("#heart", { strokeDashoffset: 0, duration: heart + .35, ease: "power1.inOut" }, 0);
tl.to("#heartCore", { strokeDashoffset: 0, duration: heart + .35, ease: "power1.inOut" }, .05);
tl.to("#inner", { strokeDashoffset: 0, duration: .5, ease: "power1.inOut" }, heart + .1);

// "beats": a strong double thump.
const beats = at("beats", .66);
const thump = (t, k) => {
  tl.to("#heartG", { scale: 1 + .09 * k, svgOrigin: "540 700", duration: .09, ease: "power2.out" }, t);
  tl.to("#heartG", { scale: 1, svgOrigin: "540 700", duration: .22, ease: "power2.in" }, t + .09);
};
thump(beats, 1.2);
thump(beats + .32, .8);

// "a hundred thousand times a day": the beats pile up, faster and faster, each leaving an echo.
const many = at("hundred", 1.26);
const fast = [0, .32, .58, .8, .98, 1.13, 1.26, 1.37];
fast.forEach((d, i) => {
  thump(many + d, .7);
  const e = "#e" + (i % 4);
  tl.fromTo(e, { opacity: .7, scale: 1, svgOrigin: "540 700" }, { opacity: 0, scale: 1.35 + i * .02, svgOrigin: "540 700", duration: .55, ease: "power2.out", immediateRender: false }, many + d);
});

// "every beat": the heart dims and its beat runs out of it as one trace across the screen.
const every = at("every", 2.9) - .1;
tl.to("#heart, #heartCore, #inner", { opacity: .35, duration: .3, ease: "power2.out" }, every);
tl.set("#ecg, #ecgCore", { opacity: 1 }, every);
tl.to("#ecg", { strokeDashoffset: 0, duration: .75, ease: "none" }, every);
tl.to("#ecgCore", { strokeDashoffset: 0, duration: .75, ease: "none" }, every + .03);

// "spike": the trace shoots up through the heart and runs on.
const spike = at("spike", 4.06) - .12;
tl.set("#spk1, #spk1Core", { opacity: 1 }, spike);
tl.to("#spk1", { strokeDashoffset: 0, duration: .6, ease: "power1.out" }, spike);
tl.to("#spk1Core", { strokeDashoffset: 0, duration: .6, ease: "power1.out" }, spike + .02);
thump(spike + .08, 1.4);
tl.to("#heart, #heartCore, #inner", { opacity: 1, duration: .1 }, spike + .08);

// "electricity": a spark cracks off the top of the spike and the screen flares.
const zap = at("electricity", 4.54) - .05;
tl.set("#bolt, #boltCore", { opacity: 1 }, zap);
tl.to("#bolt", { strokeDashoffset: 0, duration: .18, ease: "power2.out" }, zap);
tl.to("#boltCore", { strokeDashoffset: 0, duration: .18, ease: "power2.out" }, zap + .02);
tl.fromTo(".spk", { scale: 0, svgOrigin: "540 300" }, { scale: 1, svgOrigin: "540 300", duration: .25, stagger: .03, ease: "back.out(3)" }, zap + .05);
tl.to("#flash", { opacity: 1, duration: .08, yoyo: true, repeat: 3, ease: "none" }, zap);
tl.to("#bolt, #boltCore", { opacity: .4, duration: .06, yoyo: true, repeat: 5, ease: "none" }, zap + .25);
thump(zap + .4, 1);
</script>
