<script>
// The camera follows the drawing with a slow push; the board's dust drifts.
tl.fromTo("#world", { scale: 1.24 }, { scale: 1.32, transformOrigin: "540px 650px", duration: DURATION, ease: "none" }, 0);
tl.fromTo(".smudge", { x: 0 }, { x: 40, duration: DURATION, ease: "none" }, 0);
const draw = (sel, t, d, ease) => { tl.set(sel, { opacity: 1 }, t); tl.to(sel, { strokeDashoffset: 0, duration: d, ease: ease || "power1.inOut" }, t); };

// Hook: the leaf outline is sketched on straight away, then its rib and veins — "photosynthesis is simple".
draw("#leafLine", 0, .9);
draw("#rib", .6, .45);
tl.add(() => {}, 0);
document.querySelectorAll(".vein").forEach((v, i) => draw(v, .85 + i * .09, .22, "power2.out"));
// …and hatched in, a few strokes at a time.
tl.to(".h", { opacity: .55, duration: .05, stagger: .022, ease: "none" }, 1.0);

// "sunlight": the sun is drawn and its arrow swoops into the leaf.
const sun = at("sunlight", 2.86) - .45;
draw("#sunC", sun, .35);
document.querySelectorAll("#sun .ray").forEach((r, i) => draw(r, sun + .25 + i * .04, .12, "power2.out"));
draw("#sunArrow", sun + .45, .35, "power2.out");

// "water": the drop and its arrow.
const water = at("water", 3.62) - .35;
draw("#drop", water, .3);
draw("#waterArrow", water + .25, .3, "power2.out");

// "air": three breaths of wind and an arrow.
const air = at("air", 4.2) - .3;
document.querySelectorAll(".wave").forEach((w, i) => draw(w, air + i * .08, .25, "power1.out"));
draw("#airArrow", air + .3, .3, "power2.out");

// "turns them into sugar": the leaf brightens and a sugar cube grows out of its heart.
const turn = at("turns", 4.98) - .1;
tl.to(".h", { stroke: "#F6D55C", opacity: .7, duration: .3, stagger: .004, ease: "power1.out" }, turn);
tl.to("#sun, #water, #air", { opacity: .35, duration: .4 }, turn);
tl.to("#leaf", { scale: 1.04, svgOrigin: "540 650", duration: .3, yoyo: true, repeat: 1, ease: "power2.out" }, turn);
const sugar = at("sugar", 5.76) - .35;
tl.fromTo("#sugar", { scale: 0, rotation: -15, svgOrigin: "540 690" }, { scale: 1.3, rotation: 0, svgOrigin: "540 690", duration: .5, ease: "back.out(1.8)" }, sugar);
tl.fromTo(".spark", { scale: 0, opacity: 0, transformOrigin: "50% 50%" }, { scale: 1, opacity: 1, duration: .25, stagger: .05, ease: "back.out(3)", immediateRender: false }, sugar + .35);
tl.set(".spark", { opacity: 0 }, 0);
</script>
