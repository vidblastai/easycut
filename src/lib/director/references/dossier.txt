// Typewriter: one span per letter, each struck with a slightly different weight and baseline, like a real machine.
function typeOut(id, t, per) {
  const el = document.getElementById(id);
  const letters = [...el.textContent];
  el.innerHTML = letters.map((c, i) => '<span class="ch" style="position:relative; top:' + ((rand(i + id.length * 9) - 0.5) * 3).toFixed(1) + 'px; opacity:' + (0.82 + rand(i + 40) * 0.18).toFixed(2) + '">' + (c === ' ' ? '&nbsp;' : c) + '</span>').join('');
  [...el.querySelectorAll('.ch')].forEach((s, i) => tl.set(s, { display: 'inline' }, t + i * per));
  return t + letters.length * per;
}
// A stamp slams: big and off-angle to down hard, the card jolts, the camera feels it.
function slam(id, card, t, rot) {
  tl.fromTo(id, { opacity: 0, scale: 1.9, rotation: rot - 10 }, { opacity: 0.92, scale: 1, rotation: rot, duration: 0.16, ease: 'power4.in', immediateRender: false }, t);
  tl.to(card, { y: '+=6', duration: 0.05, yoyo: true, repeat: 1, ease: 'power1.out' }, t + 0.16);
  tl.to('#cam', { y: 4, duration: 0.05, yoyo: true, repeat: 1 }, t + 0.16);
}
const prep = (id) => { const el = document.getElementById(id); el.setAttribute('pathLength', '1'); el.setAttribute('stroke-dasharray', '1'); el.setAttribute('stroke-dashoffset', '1'); };
['r1', 'r2', 'r3', 'circ'].forEach(prep);
// The camera slides along the board; whatever it leaves falls slightly out of focus.
function move(x, t, soften) {
  tl.to('#board', { x: -x, duration: 1.0, ease: 'power3.inOut' }, t);
  tl.to('#lamp', { x: x * 0.25, duration: 1.4, ease: 'sine.inOut' }, t);
  if (soften) tl.to(soften, { filter: 'blur(3px)', opacity: 0.75, duration: 0.8 }, t + 0.2);
}
function pinIn(card, pin, t) {
  tl.fromTo(card, { opacity: 0, y: -50, scale: 1.04 }, { opacity: 1, y: 0, scale: 1, duration: 0.45, ease: 'power3.out', immediateRender: false }, t);
  tl.fromTo(pin, { opacity: 0, scale: 1.8 }, { opacity: 1, scale: 1, duration: 0.18, ease: 'power3.in', immediateRender: false }, t + 0.3);
}
tl.to('#cam', { scale: 1.06, duration: DURATION, ease: 'none' }, 0);
tl.set(['#b1', '#b2', '#b3'], { scaleX: 0 }, 0);

// "In March" — the date card types out.
tl.fromTo('#c1', { opacity: 0, y: -30, rotation: -9 }, { opacity: 1, y: 0, rotation: -5, duration: 0.5, ease: 'power3.out', immediateRender: false }, 0);
tl.fromTo('#p1', { opacity: 0, scale: 1.8 }, { opacity: 1, scale: 1, duration: 0.18, ease: 'power3.in', immediateRender: false }, 0.3);
typeOut('tx1', at('March', 0.08), 0.07);
// "the company lost its biggest client" — string runs to the client file; it types; LOST slams down.
tl.to('#r1', { strokeDashoffset: 0, duration: 0.6, ease: 'power2.inOut' }, at('company', 0.72) + 0.05);
pinIn('#c2', '#p2', at('company', 0.72) - 0.35);
typeOut('tx2', at('lost', 1.0) + 0.05, 0.045);
slam('#s2', '#c2', at('client', 1.86) + 0.1, -12);

// "Two weeks later, the founder resigned" — slide along the string past the tape label to the founder's file.
const t2 = at('Two', 2.58) - 0.35;
move(560, t2, '#c1');
pinIn('#c3', '#p3', t2 + 0.15);
tl.to('#r2', { strokeDashoffset: 0, duration: 0.8, ease: 'power2.inOut' }, t2 + 0.45);
tl.fromTo('#tape', { opacity: 0, scale: 1.2 }, { opacity: 1, scale: 1, duration: 0.3, ease: 'back.out(2)', immediateRender: false }, at('weeks', 2.64));
typeOut('tx3', at('founder', 3.68) - 0.05, 0.05);
slam('#s3', '#c3', at('resigned', 3.92) + 0.12, 9);

// "and nobody would say why" — on to the memo: its lines black out one by one; only WHY? is left, circled in red.

move(1060, t3, ['#c2', '#tape']);

tl.to('#r3', { strokeDashoffset: 0, duration: 0.7, ease: 'power2.inOut' }, t3 + 0.45);
['#b1', '#b2', '#b3'].forEach((b, i) => tl.to(b, { scaleX: 1, duration: 0.22, ease: 'power2.out' }, at('nobody', 4.9) + i * 0.2));
typeOut('tx4', at('why', 5.76) - 0.2, 0.05);

tl.to('#circ', { strokeDashoffset: 0, duration: 0.45, ease: 'power2.inOut' }, at('why', 5.76) + 0.08);
