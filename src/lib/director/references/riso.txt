      <pattern id="dotsYellow" width="20" height="20" patternUnits="userSpaceOnUse"><circle cx="10" cy="10" r="6" fill="#FFE800"/></pattern>
    </defs>

    <!-- the halftone field behind everything -->

      <g fill="#0078BF" class="mul">
        <rect x="532" y="250" width="16" height="46"/><rect x="532" y="824" width="16" height="46"/><rect x="250" y="552" width="46" height="16"/><rect x="784" y="552" width="46" height="16"/>

<script>
// Printed layers slide; a slight push.
tl.fromTo("#world", { scale: 1.04 }, { scale: 1.1, transformOrigin: "540px 620px", duration: DURATION, ease: "none" }, 0);
tl.fromTo("#field", { scale: .85, svgOrigin: "540 620" }, { scale: 1.05, svgOrigin: "540 620", duration: DURATION, ease: "none" }, 0);

// Hook: the cup slams onto the page and its plates slide into registration — "coffee".
tl.fromTo("#cup", { y: 500, rotation: -8, svgOrigin: "540 1010" }, { y: 0, rotation: 0, svgOrigin: "540 1010", duration: .38, ease: "expo.out" }, 0);
tl.fromTo("#cupBlue", { x: 40, y: -30 }, { x: 9, y: 7, duration: .35, ease: "power3.out" }, .2);
tl.fromTo("#clock", { scale: 0, svgOrigin: "540 560" }, { scale: 1, svgOrigin: "540 560", duration: .45, ease: "back.out(1.5)" }, .1);

// "hits your brain": steam shoots up and becomes a bolt that slams up out of the cup.
const hit = at("hits", .66) - .15;
tl.to(".st", { strokeDashoffset: 0, duration: .3, stagger: .05, ease: "power2.out" }, hit);
tl.to(".st", { opacity: 0, duration: .15 }, hit + .4);
tl.set("#boltG", { opacity: 1 }, hit + .35);
tl.set("#boltScale", { transformOrigin: "50% 100%" }, 0);
const pop = (t, d, s) => {
  tl.fromTo("#boltMove", { y: 300 }, { y: 0, duration: d, ease: "back.out(2)", immediateRender: false }, t);
  tl.fromTo("#boltScale", { scale: .2 }, { scale: s, duration: d, ease: "back.out(2)", immediateRender: false }, t);
};
pop(hit + .35, .3, 1);
tl.fromTo("#field", { opacity: .55 }, { opacity: .9, duration: .12, yoyo: true, repeat: 1, immediateRender: false }, hit + .45);

// "in ten minutes": the blue wedge sweeps a sixth of the disc.
const ten = at("10", 1.48) - .1;
const sweep = { a: 0 };
const wedge = (deg) => {
  const r = 330, cx = 540, cy = 560, rad = (deg - 90) * Math.PI / 180;
  const x = cx + r * Math.cos(rad), y = cy + r * Math.sin(rad);
  return "M540 560 L540 230 A330 330 0 " + (deg > 180 ? 1 : 0) + " 1 " + x.toFixed(1) + " " + y.toFixed(1) + " Z";
};
tl.to(sweep, { a: 60, duration: .5, ease: "power2.out", onUpdate: () => document.getElementById("wedge").setAttribute("d", wedge(sweep.a)) }, ten);

// "the crash comes back": the bolt breaks and its pieces fall into the cup, which slumps.
const crash = at("crash", 2.6) - .1;
tl.set("#boltG", { opacity: 0 }, crash);
tl.set("#crashG", { opacity: 1 }, crash);
document.querySelectorAll(".pc").forEach((p, i) => {
  tl.fromTo(p, { y: 0, rotation: 0, transformOrigin: "50% 50%" }, { y: 420 + i * 40, rotation: [-50, 70, -30][i], duration: .45, ease: "power3.in", immediateRender: false }, crash + i * .05);
});
tl.set("#crashG", { opacity: 0 }, crash + .6);
tl.to("#cup", { scaleY: .9, scaleX: 1.06, svgOrigin: "540 1010", duration: .12, yoyo: true, repeat: 1, ease: "power2.out" }, crash + .45);
tl.to("#cupBlue", { x: 22, y: 16, duration: .1, yoyo: true, repeat: 1, ease: "none" }, crash + .45);

// "an hour later": the wedge sweeps the whole way round.
const hour = at("hour", 3.66) - .3;
tl.to(sweep, { a: 359.9, duration: .55, ease: "power2.inOut", onUpdate: () => document.getElementById("wedge").setAttribute("d", wedge(sweep.a)) }, hour);

// "every single time": a loop arrow wraps the clock, and the bolt fires and crashes again, faster.
const every = at("every", 4.16) - .15;
tl.set("#loop", { opacity: 1 }, every);
tl.fromTo("#loop", { rotation: -120, scale: .8, svgOrigin: "540 560" }, { rotation: 0, scale: 1, svgOrigin: "540 560", duration: .45, ease: "back.out(1.6)", immediateRender: false }, every);
tl.to("#loop", { rotation: 40, svgOrigin: "540 560", duration: DURATION - every - .45, ease: "none" }, every + .45);
tl.set("#boltG", { opacity: 1 }, every + .2);
pop(every + .2, .22, 1);
tl.to("#boltMove", { y: 340, duration: .22, ease: "power3.in" }, every + .55);
tl.to("#boltScale", { scale: .4, duration: .22, ease: "power3.in" }, every + .55);
pop(at("time", 4.8) - .05, .25, 1.08);
</script>
