<script>
const slap = (id, t) => {
  tl.set(id, { opacity: 1 }, t);
  tl.fromTo(id, { scale: 1.45, rotation: -10, svgOrigin: "540 700" }, { scale: 1, rotation: 3, svgOrigin: "540 700", duration: .28, ease: "back.out(2.5)", immediateRender: false }, t);
  tl.to(id, { rotation: -2, svgOrigin: "540 700", duration: .08, yoyo: true, repeat: 1, ease: "sine.inOut" }, t + .28);
};
const peel = (id, t) => tl.to(id, { scaleX: 0, rotation: 12, svgOrigin: "540 700", duration: .16, ease: "power3.in" }, t);
const mark = (sel, t, d) => { tl.set(sel, { opacity: 1 }, t); tl.fromTo(sel, { strokeDashoffset: 1 }, { strokeDashoffset: 0, duration: d || .25, ease: "power2.out", immediateRender: false }, t); };
document.querySelectorAll(".mk").forEach((m) => { if (!m.getAttribute("stroke-dasharray")) { m.setAttribute("pathLength", "1"); m.setAttribute("stroke-dasharray", "1"); } });
document.querySelectorAll("#spark path").forEach((m) => { m.setAttribute("pathLength", "1"); m.setAttribute("stroke-dasharray", "1"); });

// Locked, centred; the blobs drift a touch.
tl.fromTo("#b1", { x: 0 }, { x: 40, duration: DURATION, ease: "none" }, 0);
tl.fromTo("#b2", { x: 0 }, { x: -40, duration: DURATION, ease: "none" }, 0);

// Hook: a smiling sun sticker slaps on — "my morning routine" — and gets underlined on "simple".
slap("#sun", 0);
tl.to("#sun", { rotation: 360, svgOrigin: "540 640", duration: 1.6, ease: "power1.inOut" }, .3);
mark("#under", at("simple", 1.56) - .25, .25);

// "coffee first": the sun peels off and a coffee sticker slaps on; its steam doodles on.
const coffee = at("coffee", 1.8) - .15;
peel("#sun", coffee - .1);
tl.to("#under", { opacity: 0, duration: .1 }, coffee - .1);
slap("#coffee", coffee + .05);
document.querySelectorAll("#steam .mk").forEach((m, i) => mark(m, coffee + .35 + i * .08, .25));
tl.to("#world", { scale: 1.08, transformOrigin: "540px 700px", duration: .12, yoyo: true, repeat: 1, ease: "power2.out" }, at("first", 2.38) - .05);

// "a ten minute walk": coffee peels, a sneaker slaps on and a dashed trail doodles under it.
const walk = at("10", 3.24) - .15;
peel("#coffee", walk - .1);
slap("#shoe", walk + .05);
mark("#trail", walk + .3, .6);
tl.to("#shoeIn", { y: -40, rotation: -6, svgOrigin: "540 790", duration: .15, yoyo: true, repeat: 3, ease: "power2.out" }, at("walk", 3.8) - .1);

// "only then I open my emails": the envelope slaps on, its flap flips open and the letter rises.
const mail = at("only", 4.6) - .1;
peel("#shoe", mail - .1);
slap("#mail", mail + .05);
mark("#arrow", at("then", 4.78), .25);
tl.to("#flap", { scaleY: -1, svgOrigin: "540 540", duration: .25, ease: "power3.out" }, at("open", 5.2) - .05);
tl.to("#letter", { y: -230, duration: .35, ease: "back.out(2)" }, at("open", 5.2) + .1);
tl.set("#badge", { opacity: 1 }, 0);
tl.fromTo("#badge", { scale: 0, svgOrigin: "760 540" }, { scale: 1, svgOrigin: "760 540", duration: .3, ease: "back.out(3)" }, at("emails", 5.6) - .1);
tl.set("#badge", { scale: 0, svgOrigin: "760 540" }, 0);

// Payoff: the coffee and the sneaker come back small and get ticked off, sparks round the mail.
const done = at("emails", 5.6) + .1;
tl.set("#coffee, #shoe", { scaleX: 1, rotation: 0, svgOrigin: "540 700" }, done - .01);
tl.fromTo("#coffee", { scale: 0, x: -300, y: -470, svgOrigin: "540 700" }, { scale: .42, svgOrigin: "540 700", duration: .3, ease: "back.out(2.5)", immediateRender: false }, done);
tl.fromTo("#shoe", { scale: 0, x: 300, y: -470, svgOrigin: "540 700" }, { scale: .42, svgOrigin: "540 700", duration: .3, ease: "back.out(2.5)", immediateRender: false }, done + .08);
mark("#tick1", done + .3, .2);
mark("#tick2", done + .4, .2);
document.querySelectorAll("#spark path").forEach((m, i) => mark(m, done + .2 + i * .05, .15));
</script>
