<script>
const draw = (el, t, d, ease) => { tl.set(el, { opacity: 1 }, t); tl.to(el, { strokeDashoffset: 0, duration: d, ease: ease || "power1.inOut" }, t); };
// The draftsman's eye: a steady, slow push into the drawing.
tl.fromTo("#world", { scale: 1.02 }, { scale: 1.1, transformOrigin: "540px 660px", duration: DURATION, ease: "none" }, 0);

// Hook: guide lines sweep across, then the rim draws round on "bicycle wheel".
tl.set("#guides .d", { opacity: 1 }, 0);
tl.fromTo("#gH", { scaleX: 0, svgOrigin: "540 660" }, { scaleX: 1, svgOrigin: "540 660", duration: .35, ease: "power2.out", immediateRender: false }, 0);
tl.fromTo("#gV", { scaleY: 0, svgOrigin: "540 660" }, { scaleY: 1, svgOrigin: "540 660", duration: .35, ease: "power2.out", immediateRender: false }, .08);
tl.fromTo("#gC", { scale: 0, svgOrigin: "540 660" }, { scale: 1, svgOrigin: "540 660", duration: .4, ease: "power2.out", immediateRender: false }, .15);
draw("#rimO", .15, .7);
draw("#rimI", .3, .65);
draw("#tyre", .45, .6);
tl.set("#hidden", { opacity: .55 }, .9);
tl.set("#hub", { opacity: 1 }, at("wheel", .86));
tl.fromTo("#hub", { scale: 0, svgOrigin: "540 660" }, { scale: 1, svgOrigin: "540 660", duration: .3, ease: "power3.out", immediateRender: false }, at("wheel", .86));
tl.to("#guides", { opacity: .4, duration: .4 }, 1.2);

// "stays strong": the dimension line extends from its ticks.
const strong = at("strong", 1.68) - .1;
tl.set("#dim", { opacity: 1 }, strong);
tl.fromTo("#dim", { scaleX: 0, svgOrigin: "540 1060" }, { scaleX: 1, svgOrigin: "540 1060", duration: .4, ease: "power2.out", immediateRender: false }, strong);

// "thirty six thin spokes": laced in, one after another, all the way round.
const spokes = at("36", 2.44) - .1;
document.querySelectorAll(".spoke").forEach((s, i) => draw(s, spokes + i * .028, .22, "power1.out"));
tl.set("#nipples", { opacity: 1 }, spokes + 1);
tl.fromTo("#nipples circle", { scale: 0, transformOrigin: "50% 50%" }, { scale: 1, duration: .15, stagger: .008, ease: "back.out(3)", immediateRender: false }, spokes + 1);

// "pull on the rim": tension arrows press in; "from every side": they come from all round.
const pull = at("pull", 3.66) - .1;
for (let i = 0; i < 12; i++) {
  const t = i < 2 ? pull + i * .1 : at("every", 4.74) - .3 + (i - 2) * .05;
  tl.set(".a" + i, { opacity: 1 }, t);
  tl.fromTo(".a" + i, { scale: 1.35, svgOrigin: "540 660" }, { scale: 1, svgOrigin: "540 660", duration: .3, ease: "power3.out", immediateRender: false }, t);
}
tl.to("#rimO, #rimI", { stroke: "#FFC94A", duration: .2, ease: "power1.out" }, at("rim", 4.3) - .1);
tl.to(".spoke", { stroke: "#FFC94A", duration: .15, stagger: .01, ease: "none" }, at("rim", 4.3));

// Payoff on "side": the arrows squeeze in once together and the finished wheel turns.
const side = at("side", 5.04);
tl.to("#arrows", { scale: .97, svgOrigin: "540 660", duration: .12, yoyo: true, repeat: 1, ease: "power2.out" }, side);
tl.to("#spokes, #nipples", { rotation: 20, svgOrigin: "540 660", duration: DURATION - side, ease: "power1.out" }, side - .2);
</script>
