<script>
// Paper on a table: a gentle push and a slight slide across the layers.
tl.fromTo("#world", { scale: 1.12, x: 10 }, { scale: 1.2, x: -10, transformOrigin: "540px 680px", duration: DURATION, ease: "none" }, 0);
tl.fromTo("#sh1", { x: 0 }, { x: 30, duration: DURATION, ease: "none" }, 0);
tl.fromTo("#sh2", { x: 0 }, { x: -24, duration: DURATION, ease: "none" }, 0);
// Wings flutter all the way through (finite).
tl.to("#wingF, #wingB", { scaleY: .55, transformOrigin: "50% 100%", duration: .06, yoyo: true, repeat: 61, ease: "none" }, 0);

// Hook: the flower folds up like a pop-up page while the bee flies in on "bee".
tl.fromTo("#flower", { scaleY: 0, svgOrigin: "540 980" }, { scaleY: 1, svgOrigin: "540 980", duration: .5, ease: "back.out(1.4)" }, 0);
tl.fromTo("#petals", { rotation: -25, svgOrigin: "540 630" }, { rotation: 0, svgOrigin: "540 630", duration: .6, ease: "back.out(1.6)" }, .15);
const bee = at("bee", .76) - .55;
tl.fromTo("#bee", { x: -160, y: 760, rotation: -20 }, { x: 300, y: 600, rotation: 8, duration: .5, ease: "power2.out" }, Math.max(0, bee));
// "visits": it lands on the flower's heart with a little squash.
const visit = at("visits", 1.02);
tl.to("#bee", { x: 540, y: 560, rotation: 0, duration: .35, ease: "back.out(1.6)" }, visit);
tl.set("#trail", { opacity: .5 }, Math.max(0, bee));
tl.fromTo("#trail", { strokeDashoffset: 400 }, { strokeDashoffset: 0, duration: .9, ease: "power1.out" }, Math.max(0, bee));
tl.to("#trail", { opacity: 0, duration: .3 }, visit + .5);
tl.to("#flower", { scaleY: .94, svgOrigin: "540 980", duration: .1, yoyo: true, repeat: 1 }, visit + .3);

// "five thousand flowers": a ring of flowers pops up around it and the bee zips between them.
const many = at("5", 1.52) - .1;
tl.set(".ring", { opacity: 1 }, many);
document.querySelectorAll(".ring").forEach((r, i) => {
  tl.fromTo(r.firstElementChild, { scale: 0, rotation: -60 }, { scale: 1, rotation: 0, duration: .35, ease: "back.out(2)", immediateRender: false }, many + i * .07);
});
const stops = [[230, 450], [860, 450], [180, 690], [910, 690], [540, 560]];
stops.forEach((p, i) => {
  tl.to("#bee", { x: p[0], y: p[1], rotation: i % 2 ? 12 : -12, scale: .8, duration: .16, ease: "power2.inOut" }, at("flowers", 2.38) - .3 + i * .17);
});
tl.to("#bee", { scale: 1, rotation: 0, duration: .2 }, at("flowers", 2.38) - .3 + 5 * .17);

// "every trip makes the honey": the flowers fold down flat and a paper jar flips up in their place.
const trip = at("every", 3.48) - .15;
tl.to(".ring", { scaleY: 0, transformOrigin: "50% 100%", duration: .22, stagger: .02, ease: "power3.in" }, trip);
tl.to("#flower", { scaleY: 0, svgOrigin: "540 980", duration: .25, ease: "power3.in" }, trip);
tl.set("#jar", { opacity: 1 }, trip + .22);
tl.fromTo("#jar", { scaleY: 0, rotation: 6, svgOrigin: "540 990" }, { scaleY: 1, rotation: 0, svgOrigin: "540 990", duration: .45, ease: "back.out(1.5)", immediateRender: false }, trip + .22);
tl.to("#bee", { x: 760, y: 420, rotation: 10, duration: .3, ease: "power2.out" }, trip + .1);
// …and every trip into the jar raises the honey a step.
for (let k = 0; k < 3; k++) {
  const t = at("honey", 4.38) - .35 + k * .3;
  tl.to("#bee", { x: 600, y: 470, rotation: -20, duration: .12, ease: "power2.in" }, t);
  tl.to("#bee", { x: 770, y: 400, rotation: 10, duration: .14, ease: "power2.out" }, t + .12);
  tl.to("#honey", { y: -110 * (k + 1), duration: .25, ease: "back.out(1.6)" }, t + .1);
}
// "a little sweeter": the jar is full, honey drips over the rim and the jar does a happy hop.
const sweet = at("sweeter", 5.08) - .1;
tl.to("#honey", { y: -400, duration: .3, ease: "back.out(1.4)" }, sweet);
tl.set("#drip", { opacity: 1 }, sweet + .15);
tl.fromTo("#drip", { scaleY: 0, svgOrigin: "700 520" }, { scaleY: 1, svgOrigin: "700 520", duration: .4, ease: "back.out(1.6)", immediateRender: false }, sweet + .15);
tl.to("#jar", { y: -30, duration: .14, yoyo: true, repeat: 1, ease: "power2.out" }, sweet + .1);
tl.to("#bee", { x: 540, y: 380, rotation: 0, scale: 1.1, duration: .35, ease: "back.out(1.6)" }, sweet + .2);
</script>
