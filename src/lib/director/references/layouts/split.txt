<script>
// The colour behind the glass drifts the whole time.
tl.fromTo("#b1", { x: 0, y: 0 }, { x: 80, y: 40, duration: DURATION, ease: "sine.inOut" }, 0);
tl.fromTo("#b2", { x: 0 }, { x: -90, duration: DURATION, ease: "sine.inOut" }, 0);
tl.fromTo("#b3", { x: 0, y: 0 }, { x: 60, y: -40, duration: DURATION, ease: "sine.inOut" }, 0);
const sweep = { a: 0 };
const wedge = () => {
  const deg = sweep.a % 360.0001, r = 118, rad = (deg - 90) * Math.PI / 180;
  const x = 525 + r * Math.cos(rad), y = 430 + r * Math.sin(rad);
  document.getElementById("eaten").setAttribute("d", "M525 430 L525 312 A118 118 0 " + (deg > 180 ? 1 : 0) + " 1 " + x.toFixed(1) + " " + y.toFixed(1) + " Z");
};

tl.fromTo("#bottom", { y: 160, opacity: 0 }, { y: 0, opacity: .3, duration: .6, ease: "expo.out" }, .2);

// Hook: the top panel glides in with its week of tiles and the clock already ticking — "editing used to".
tl.fromTo("#top", { y: 120, opacity: 0, scale: .9, transformOrigin: "50% 50%" }, { y: 0, opacity: 1, scale: 1, transformOrigin: "50% 50%", duration: .5, ease: "expo.out" }, 0);
tl.fromTo(".tile", { y: 60, opacity: 0 }, { y: 0, opacity: 1, duration: .35, stagger: .05, ease: "expo.out" }, .15);
tl.fromTo("#clock", { scale: .6, opacity: 0, svgOrigin: "525 430" }, { scale: 1, opacity: 1, svgOrigin: "525 430", duration: .45, ease: "expo.out" }, .1);
tl.to("#hand", { rotation: 360 * 4, svgOrigin: "525 430", duration: 2.2, ease: "power1.in" }, .2);

// "eat your whole weekend": the clock's pink wedge sweeps round as a pink bite swallows the last two tiles.
const eat = at("eat", 1.08) - .1;
tl.to(sweep, { a: 359.9, duration: .9, ease: "power2.inOut", onUpdate: wedge }, eat);
tl.to("#bite", { attr: { width: 194 }, duration: .45, ease: "power2.inOut" }, at("weekend", 1.52) - .2);
tl.to(".wk", { scale: .82, opacity: .6, transformOrigin: "50% 50%", duration: .3, ease: "power2.in" }, at("weekend", 1.52));

// "now": the top dims back, the chevron points down and the bottom panel arrives.
const now = at("now", 2.34) - .15;
tl.to("#top, .tile, #clock, #bite", { opacity: .45, duration: .4, ease: "power2.out" }, now);
tl.set("#arrow", { opacity: 1 }, now);
tl.fromTo("#arrow", { y: -30 }, { y: 0, duration: .3, ease: "back.out(2)", immediateRender: false }, now);
tl.fromTo("#bottom", { y: 0, opacity: .3, scale: .94, transformOrigin: "50% 50%" }, { y: 0, opacity: 1, scale: 1, transformOrigin: "50% 50%", duration: .5, ease: "expo.out", immediateRender: false }, now + .05);
tl.set("#bottomArt", { opacity: 1 }, now + .15);
tl.fromTo("#cup", { scale: .6, svgOrigin: "420 1220" }, { scale: 1, svgOrigin: "420 1220", duration: .4, ease: "back.out(1.8)", immediateRender: false }, now + .15);

// "it's done": the ring snaps shut in a blink and the check lands.
const done = at("done", 2.76) - .1;
tl.to("#ring", { strokeDashoffset: 0, duration: .45, ease: "power3.inOut" }, done);
tl.fromTo("#check", { scale: 0, svgOrigin: "740 1190" }, { scale: 1, svgOrigin: "740 1190", duration: .4, ease: "back.out(2)" }, done + .3);
tl.to("#tick", { strokeDashoffset: 0, duration: .25, ease: "power2.out" }, done + .5);

// "before your coffee gets cold": the steam keeps curling — still hot — and the cup glows warm.
document.querySelectorAll(".sm").forEach((m, i) => {
  tl.fromTo(m, { y: 20, opacity: 0 }, { y: -40, opacity: 1, duration: .6, repeat: 3, ease: "sine.inOut", immediateRender: false }, at("coffee", 3.46) - .5 + i * .15);
});
tl.to("#cup path:first-child", { fill: "rgba(255,181,71,.45)", duration: .4 }, at("cold", 4.02) - .3);
tl.to("#bottom", { scale: 1.03, transformOrigin: "50% 50%", duration: .2, yoyo: true, repeat: 1, ease: "power2.out" }, at("cold", 4.02) - .1);
</script>
