<script>
document.querySelectorAll(".mk").forEach((m) => { if (!m.getAttribute("stroke-dasharray")) { m.setAttribute("pathLength", "1"); m.setAttribute("stroke-dasharray", "1"); m.setAttribute("stroke-dashoffset", "1"); } });
const slap = (id, t) => {
  tl.set(id, { opacity: 1 }, t);
  tl.fromTo(id + "S", { scale: 1.5, rotation: -10, transformOrigin: "50% 50%" }, { scale: 1, rotation: 3, transformOrigin: "50% 50%", duration: .28, ease: "back.out(2.5)", immediateRender: false }, t);
  tl.to(id + "S", { rotation: -1, transformOrigin: "50% 50%", duration: .08, yoyo: true, repeat: 1, ease: "sine.inOut" }, t + .28);
};
const settle = (id, t) => tl.to(id + "S", { scale: .9, transformOrigin: "50% 50%", duration: .3, ease: "power2.out" }, t);
const mark = (sel, t, d) => {
  tl.set(sel, { opacity: 1 }, t);
  if (document.querySelector(sel).getAttribute("stroke-dasharray") === "1") tl.to(sel, { strokeDashoffset: 0, duration: d || .25, ease: "power2.out" }, t);
};
tl.set("#badge", { scale: 0, transformOrigin: "50% 50%" }, 0);
tl.fromTo("#b1", { x: 0 }, { x: 50, duration: DURATION, ease: "none" }, 0);
tl.fromTo("#b2", { x: 0 }, { x: -50, duration: DURATION, ease: "none" }, 0);

// Hook: a big smiling sun slaps on — "my morning routine" — then flies up to sit small in the corner.
slap("#sun", 0);
tl.to("#sunS", { rotation: 360, transformOrigin: "50% 50%", duration: 1.4, ease: "power1.inOut" }, .3);
const simple = at("simple", 1.56) - .2;
tl.to("#sun", { attr: { transform: "translate(860 330)" }, duration: .4, ease: "power3.inOut" }, simple);
tl.to("#sunS", { scale: .38, transformOrigin: "50% 50%", duration: .4, ease: "power3.inOut" }, simple);

// Row one, "coffee first": top left, with its steam doodled on.
const coffee = at("coffee", 1.8) - .1;
slap("#coffee", coffee);
document.querySelectorAll(".st").forEach((m, i) => { tl.set(m, { opacity: 1 }, coffee + .3 + i * .08); tl.to(m, { strokeDashoffset: 0, duration: .25, ease: "power2.out" }, coffee + .3 + i * .08); });

// Row two, "a ten minute walk": a marker arrow leads down to a sneaker on the right, which walks a dashed trail.
const walk = at("10", 3.24) - .35;
mark("#link1", walk - .1, .25);
slap("#shoe", walk + .1);
settle("#coffee", walk + .1);
mark("#tick1", walk + .35, .2);
mark("#trail", at("walk", 3.8) - .15, .5);
tl.to("#shoeS", { y: -30, rotation: -6, transformOrigin: "50% 100%", duration: .14, yoyo: true, repeat: 3, ease: "power2.out" }, at("walk", 3.8) - .1);

// Row three, "only then I open my emails": the arrow leads down-left to the envelope, which opens.
const mail = at("only", 4.6) - .3;
mark("#link2", mail - .1, .25);
slap("#mail", mail + .1);
settle("#shoe", mail + .1);
mark("#tick2", mail + .35, .2);
tl.to("#flap", { scaleY: -1, svgOrigin: "540 540", duration: .25, ease: "power3.out" }, at("open", 5.2) - .05);
tl.to("#letter", { y: -230, duration: .35, ease: "back.out(2)" }, at("open", 5.2) + .1);

// Payoff on "emails": the last sticker gets a ring, sparks and a notification dot.
const done = at("emails", 5.6) - .05;
mark("#ring", done, .35);
document.querySelectorAll(".sp").forEach((m, i) => mark("#spark .sp:nth-child(" + (i + 1) + ")", done + .25 + i * .05, .15));
tl.to("#badge", { scale: 1, transformOrigin: "50% 50%", duration: .3, ease: "back.out(3)" }, done + .2);
</script>
