<script>
const road = document.getElementById("road");
const L = road.getTotalLength();
const pos = { p: 0 };
const place = () => {
  const pt = road.getPointAtLength(pos.p * L), ahead = road.getPointAtLength(Math.min(L, pos.p * L + 4));
  const tilt = Math.atan2(ahead.y - pt.y, ahead.x - pt.x) * 180 / Math.PI;
  document.getElementById("parcel").setAttribute("transform", "translate(" + pt.x.toFixed(1) + " " + (pt.y - 40).toFixed(1) + ") rotate(" + (Math.max(-25, Math.min(25, tilt * .3))).toFixed(1) + ")");
};
place();
// Move the parcel to a fraction of the road, drawing the road just ahead of it.
const ride = (to, t, d, ease) => {
  tl.to(pos, { p: to, duration: d, ease: ease || "power1.inOut", onUpdate: place }, t);
  tl.to("#road", { strokeDashoffset: Math.max(0, 1 - to - .08), duration: d, ease: ease || "power1.inOut" }, t - .1);
};
tl.fromTo("#world", { scale: 1.02 }, { scale: 1.06, transformOrigin: "540px 880px", duration: DURATION, ease: "none" }, 0);

// Hook: the warehouse pops up under the night sky and the parcel sits at its door — "your parcel".
tl.fromTo("#warehouse", { scaleY: 0, transformOrigin: "50% 100%" }, { scaleY: 1, duration: .45, ease: "back.out(1.5)" }, 0);
tl.fromTo("#moonG", { y: -200 }, { y: 0, duration: .5, ease: "back.out(1.4)" }, .1);
tl.fromTo("#parcelIn", { scale: 0, transformOrigin: "50% 50%" }, { scale: 1, transformOrigin: "50% 50%", duration: .35, ease: "back.out(2.5)" }, at("parcel", .46) - .15);
tl.set("#road, #roadDash", { opacity: 1 }, .5);
tl.fromTo("#roadDash", { opacity: 0 }, { opacity: .8, duration: .3, immediateRender: false }, .5);

// "leaves the warehouse at night": it sets off down the road.
ride(.18, at("leaves", .68), 1.2);

// "crosses three cities": each city folds up as the parcel passes it.
const cross = at("crosses", 2.12) - .1;
const stops = [.25, .5, .72];
["#city1", "#city2", "#city3"].forEach((c, i) => {
  const t = cross + .1 + i * .45;
  tl.set(c, { opacity: 1 }, t);
  tl.fromTo(c, { scaleY: 0, transformOrigin: "50% 100%" }, { scaleY: 1, transformOrigin: "50% 100%", duration: .3, ease: "back.out(1.8)", immediateRender: false }, t);
});
ride(.74, cross, 1.5, "none");
// The night sheet slides up and away as morning comes.
tl.to("#night", { y: -1300, duration: 1.6, ease: "power2.inOut" }, cross + .6);
tl.to("#moonG", { y: -900, duration: 1.4, ease: "power2.in" }, cross + .6);

// "lands on your doorstep": the house pops up and the parcel drops onto the step.
const land = at("lands", 3.72) - .2;
tl.set("#house", { opacity: 1 }, land - .15);
tl.fromTo("#house", { scaleY: 0, transformOrigin: "50% 100%" }, { scaleY: 1, transformOrigin: "50% 100%", duration: .4, ease: "back.out(1.6)", immediateRender: false }, land - .15);
ride(1, land, .55, "power2.out");
tl.to("#parcelIn", { scaleY: .82, scaleX: 1.12, transformOrigin: "50% 100%", duration: .1, yoyo: true, repeat: 1, ease: "power2.out" }, at("doorstep", 4.34));

// Payoff on "breakfast": the sun pops up beside the house.
const sun = at("breakfast", 5.02) - .2;
tl.set("#sun", { opacity: 1 }, sun);
tl.fromTo("#sun", { y: 120, scale: .6, transformOrigin: "50% 50%" }, { y: -110, scale: 1.2, transformOrigin: "50% 50%", duration: .5, ease: "back.out(1.6)", immediateRender: false }, sun);
tl.to("#house rect:nth-of-type(2)", { fill: "#F2A541", duration: .2 }, sun + .2);
</script>
