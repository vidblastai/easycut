<script>
const draw = (el, t, d, ease) => { tl.set(el, { opacity: 1 }, t); tl.to(el, { strokeDashoffset: 0, duration: d, ease: ease || "power1.inOut" }, t); };
tl.fromTo("#world", { scale: 1.08 }, { scale: 1.16, transformOrigin: "540px 640px", duration: DURATION, ease: "none" }, 0);

// Hook: a clean bulb draws itself at once — "most great ideas".
const clean = document.querySelectorAll("#clean .d");
draw(clean[0], 0, .7);
draw(clean[1], .45, .3, "power2.out");
draw(clean[2], .6, .35, "power2.out");

// "a bad sketch on a napkin": the napkin slaps down, the clean bulb lifts away and a scribble takes its place.
const sketch = at("bad", 1.72) - .35;
tl.set("#napkin", { opacity: 1 }, sketch);
tl.fromTo("#napkin", { y: 700, rotation: -14, svgOrigin: "540 630" }, { y: 0, rotation: -4, svgOrigin: "540 630", duration: .4, ease: "back.out(1.3)", immediateRender: false }, sketch);
tl.to("#clean", { opacity: 0, scale: 1.1, svgOrigin: "540 620", duration: .25, ease: "power2.in" }, sketch + .15);
document.querySelectorAll("#messy .m").forEach((m, i) => draw(m, sketch + .35 + i * .14, .28, "power2.out"));
// It boils a little — a hand-drawn line never sits perfectly still.
tl.to("#messy", { x: 2, y: -2, rotation: .6, svgOrigin: "540 620", duration: .08, yoyo: true, repeat: 11, ease: "steps(1)" }, sketch + 1);

// "stop waiting": three dots tick on, and a red stroke strikes them out.
const wait = at("waiting", 3.62) - .3;
tl.set("#wait", { opacity: 1 }, wait - .1);
tl.fromTo(".wd", { scale: 0, transformOrigin: "50% 50%" }, { scale: 1, duration: .15, stagger: .1, ease: "back.out(3)", immediateRender: false }, wait - .1);
draw("#strike", wait + .3, .2, "power3.out");

// "for the perfect one": a perfect circle starts closing round the sketch… and gives up.
const perfect = at("perfect", 4.28) - .3;
tl.to("#wait", { opacity: 0, duration: .2 }, perfect);
tl.set("#perfect", { opacity: .9 }, perfect);
tl.to("#perfect", { strokeDashoffset: .3, duration: .45, ease: "power1.inOut" }, perfect);
tl.to("#perfect", { opacity: 0, scale: 1.08, svgOrigin: "540 620", duration: .25, ease: "power2.in" }, perfect + .5);

// Payoff on "one": the scrappy sketch is the one that lights up — the only red in the picture.
const one = at("one", 4.62) + .05;
tl.to("#fill", { opacity: .9, duration: .25, ease: "power2.out" }, one);
tl.set("#glow", { opacity: 1 }, one + .05);
tl.fromTo(".ray", { scale: 0, transformOrigin: "50% 100%" }, { scale: 1, duration: .25, stagger: .04, ease: "back.out(3)", immediateRender: false }, one + .05);
tl.to("#messy", { scale: 1.05, svgOrigin: "540 620", duration: .15, yoyo: true, repeat: 1, ease: "power2.out" }, one);
</script>
