<script>
// Grain boils every frame-pair; the picture is never perfectly still.
tl.to("#grain", { x: 40, y: -30, duration: .08, repeat: Math.floor(DURATION / .08), yoyo: true, ease: "steps(1)" }, 0);
tl.fromTo("#world", { scale: 1.12 }, { scale: 1.2, transformOrigin: "540px 640px", duration: DURATION, ease: "none" }, 0);
tl.to(".gh", { y: 60, duration: .5, repeat: Math.floor(DURATION / .5), ease: "none" }, 0);

// Hook: the old screen switches on — a bright line, then the picture snaps open — "in 1985".
tl.set("#flash", { opacity: 1 }, 0);
tl.to("#flash", { scaleX: 1, opacity: 0, duration: .25, ease: "power2.in" }, .12);
tl.fromTo("#world", { scaleY: .01 }, { scaleY: 1, transformOrigin: "540px 640px", duration: .22, ease: "expo.out" }, .08);
tl.fromTo("#horizon", { opacity: .4 }, { opacity: 1, duration: .1, repeat: 3, yoyo: true, ease: "steps(1)" }, .3);

// "a home computer": it rises out of the horizon and its cursor blinks.
const comp = at("home", 1.96) - .25;
tl.set("#computer", { opacity: 1 }, comp);
tl.fromTo("#computer", { y: 700 }, { y: 0, duration: .45, ease: "back.out(1.3)", immediateRender: false }, comp);
tl.set("#fringe", { opacity: .5 }, comp);
tl.fromTo("#fringe", { y: 700 }, { y: 0, duration: .45, ease: "back.out(1.3)", immediateRender: false }, comp);
tl.to("#horizon", { opacity: .45, duration: .4 }, comp);
tl.to("#cursor", { opacity: 0, duration: .25, repeat: 9, yoyo: true, ease: "steps(1)" }, comp + .4);

// "had less memory": the bar on its screen fills… one block. That is all of it.
const mem = at("memory", 3.12) - .2;
tl.to(".blk", { opacity: 1, duration: .05 }, mem);
tl.to("#mem", { x: 3, duration: .05, yoyo: true, repeat: 5, ease: "steps(1)" }, mem + .1);

// "than one photo": the computer steps back and a polaroid pops up beside it; its bar fills and overflows.
const photo = at("photo", 3.94) - .2;
tl.to("#computer, #fringe", { scale: .62, x: -200, y: 120, svgOrigin: "540 640", duration: .35, ease: "power3.out" }, photo);
tl.set("#photo", { opacity: 1 }, photo + .05);
tl.fromTo("#photo", { scale: 0, rotation: -20, svgOrigin: "680 570" }, { scale: 1, rotation: 6, svgOrigin: "680 570", duration: .35, ease: "back.out(2)", immediateRender: false }, photo + .05);
tl.set("#pbar", { opacity: 1 }, photo + .3);
tl.to("#pfill", { attr: { width: 704 }, duration: .55, ease: "power2.in" }, photo + .3);

// "on your phone today": the picture rolls like an old TV and the photo lands as the payoff.
const today = at("today", 4.9) - .2;
tl.to("#world", { y: -30, duration: .06, yoyo: true, repeat: 3, ease: "steps(1)" }, today);
tl.to("#photo", { scale: 1.12, rotation: 3, svgOrigin: "680 570", duration: .4, ease: "back.out(2)" }, today + .15);
tl.to("#pfill", { fill: "#FFE8C2", duration: .08, yoyo: true, repeat: 5, ease: "steps(1)" }, today + .1);
</script>
