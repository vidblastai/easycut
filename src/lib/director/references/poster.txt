<script>
tl.set("#figure", { scale: 1.35, transformOrigin: "540px 560px" }, 0);

// Hook: three bars slam in on the beat, one per habit.
["#bar1", "#bar2", "#bar3"].forEach((id, i) => {
  tl.fromTo(id, { x: i % 2 ? 1100 : -1100 }, { x: 0, duration: .32, ease: "expo.out" }, .05 + i * .16);
});
tl.fromTo(".dot", { scale: 0 }, { scale: 1, duration: .25, stagger: .16, ease: "back.out(3)" }, .3);
// "changed everything": the stack punches.
const all = at("everything", 1.08);
tl.to(".bar", { scaleX: 1.08, duration: .1, yoyo: true, repeat: 1, stagger: .05, ease: "power2.out" }, all);
tl.to("#far", { scale: 1.12, transformOrigin: "50% 50%", duration: 2, ease: "power1.out" }, all);

// "wake up early": the bars snap away and a sun rises over a horizon.
const wake = at("wake", 1.98) - .25;
tl.to("#bar1", { scaleX: 0, transformOrigin: "100% 50%", duration: .22, ease: "power3.in" }, wake);
tl.to("#bar2", { scaleX: 0, transformOrigin: "0% 50%", duration: .22, ease: "power3.in" }, wake + .05);
tl.to("#bar3", { scaleY: .43, y: -10, duration: .22, ease: "power3.in" }, wake + .05);
tl.set("#horizon, #sunBox", { opacity: 1 }, wake + .25);
tl.set("#bar3", { opacity: 0 }, wake + .28);
tl.fromTo("#sun", { y: 0 }, { y: -380, duration: .55, ease: "back.out(1.3)", immediateRender: false }, wake + .28);
tl.fromTo(".ray", { scaleY: 0, opacity: 1 }, { scaleY: 1, opacity: 1, duration: .25, stagger: .05, ease: "back.out(2.5)", immediateRender: false }, at("early", 2.3) - .05);
tl.set(".ray", { opacity: 0 }, 0);

// "move your body": the sun wipes off and a body of shapes snaps through moves.
const move = at("move", 2.94) - .15;
tl.to("#sunBox, #horizon, #rays", { x: -1200, duration: .25, ease: "power3.in" }, move);
tl.set("#figure", { opacity: 1 }, move + .2);
tl.fromTo("#figure", { x: 1100 }, { x: 0, duration: .3, ease: "expo.out", immediateRender: false }, move + .2);
const poses = [[-45, 225, 70, 110], [-135, 315, 40, 140], [-45, 225, 100, 80], [-150, 330, 60, 120]];
poses.forEach((p, i) => {
  const t = move + .35 + i * .2;
  tl.to("#armR", { rotation: p[0], duration: .1, ease: "power4.out" }, t);
  tl.to("#armL", { rotation: p[1], duration: .1, ease: "power4.out" }, t);
  tl.to("#legR", { rotation: p[2], duration: .1, ease: "power4.out" }, t);
  tl.to("#legL", { rotation: p[3], duration: .1, ease: "power4.out" }, t);
  tl.to("#figure", { y: i % 2 ? 0 : -40, duration: .1, ease: "power3.out" }, t);
});

// "finish one hard task": the body slides out, a pink plate lands and a giant check slams onto it.
const finish = at("finish", 3.82) - .1;
tl.to("#figure", { x: 1200, duration: .22, ease: "power3.in" }, finish);
tl.set("#plate", { opacity: 1 }, finish + .15);
tl.fromTo("#plate", { scale: 0 }, { scale: 1, duration: .35, ease: "back.out(1.6)", immediateRender: false }, finish + .15);
const hard = at("hard", 4.34) - .05;
tl.set("#check", { opacity: 1 }, hard);
tl.fromTo("#c1", { scaleX: 0 }, { scaleX: 1, duration: .14, ease: "power4.out", immediateRender: false }, hard);
tl.fromTo("#c2", { scaleX: 0 }, { scaleX: 1, duration: .2, ease: "power4.out", immediateRender: false }, hard + .14);
tl.fromTo("#check", { scale: 1.25 }, { scale: 1, transformOrigin: "540px 700px", duration: .3, ease: "expo.out", immediateRender: false }, hard + .34);

// "before lunch": the payoff — the plate punches and three shapes burst out around it.
const lunch = at("lunch", 5.28) - .15;
tl.to("#plate", { scale: 1.08, duration: .12, yoyo: true, repeat: 1, ease: "power2.out" }, lunch);
tl.set(".conf", { opacity: 1 }, lunch);
tl.fromTo("#k1", { scale: 0, rotation: -90 }, { scale: 1, rotation: 0, duration: .3, ease: "back.out(2.5)", immediateRender: false }, lunch);
tl.fromTo("#k2", { scale: 0 }, { scale: 1, duration: .3, ease: "back.out(2.5)", immediateRender: false }, lunch + .06);
tl.fromTo("#k3", { scale: 0, rotation: 90 }, { scale: 1, rotation: 0, duration: .3, ease: "back.out(2.5)", immediateRender: false }, lunch + .12);
</script>
