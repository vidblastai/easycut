<script>
// A straight, level push into the set — never tilted.
tl.fromTo("#world", { scale: 1.12 }, { scale: 1.2, transformOrigin: "540px 640px", duration: DURATION, ease: "none" }, 0);

// Hook: the hill is there and the town pops up behind it in mirrored pairs on "Lisbon".
const lisbon = Math.min(at("Lisbon", .5), .5) - .4;
tl.fromTo("#hill", { y: 120, opacity: 0 }, { y: 0, opacity: 1, duration: .45, ease: "power3.out" }, 0);
tl.fromTo("#arch", { scaleY: 0, svgOrigin: "540 1000" }, { scaleY: 1, duration: .6, ease: "power3.out" }, .1);
[["#h2", "#h5"], ["#h1", "#h4"], ["#h0", "#h3"]].forEach((pair, i) => {
  tl.fromTo(pair, { y: 520 }, { y: 0, duration: .5, ease: "back.out(1.4)" }, lisbon + .3 + i * .12);
});

// "tiny yellow tram": the tram slides in at the foot of the hill, already on the slope.
const tram = at("tram", 1.78) - .45;
const footX = 230, footY = 910;
tl.set("#track, #ties", { opacity: 1 }, tram - .2);
tl.to("#track", { strokeDashoffset: 0, duration: .6, ease: "power2.inOut" }, tram - .2);
tl.fromTo("#ties line", { scale: 0, transformOrigin: "50% 50%" }, { scale: 1, duration: .2, stagger: .05, ease: "back.out(3)" }, tram);
tl.set("#tram", { opacity: 1 }, tram);
tl.fromTo("#tram", { x: footX - 240, y: footY + 240, rotation: -45, scale: 1.25 }, { x: footX, y: footY, scale: 1.25, duration: .45, ease: "power3.out", immediateRender: false }, tram);
tl.to("#lamp", { fill: "#E7B75F", duration: .1, yoyo: true, repeat: 3, ease: "none" }, tram + .45);

// "climbs the steepest hill": up the 45° slope, then it levels out on the plateau at the top.
const climb = at("climbs", 2.08) - .05;
tl.to("#tram", { x: 470, y: 670, duration: 1.15, ease: "power1.inOut" }, climb);
tl.to("#tram", { x: 540, y: 626, rotation: 0, scale: 1.3, duration: .5, ease: "power2.out" }, climb + 1.15);
for (let i = 0; i < 3; i++) {
  const t = climb + .2 + i * .4;
  tl.fromTo(".puff:nth-child(" + (i + 1) + ")", { x: 230 + i * 80, y: 820 - i * 80, scale: .2, opacity: .9 }, { x: 190 + i * 80, y: 760 - i * 80, scale: 1.4, opacity: 0, duration: .7, ease: "power1.out", immediateRender: false }, t);
}

// "every twelve minutes": the clock pops in above and its minute hand sweeps a full turn.
const every = at("every", 3.86) - .2;
tl.fromTo("#clock", { scale: 0, svgOrigin: "540 330" }, { scale: 1, svgOrigin: "540 330", duration: .5, ease: "back.out(1.8)" }, every);
tl.fromTo("#minHand", { rotation: 0, svgOrigin: "540 330" }, { rotation: 360, svgOrigin: "540 330", duration: 1.1, ease: "power2.inOut" }, every + .35);
tl.fromTo("#hourHand", { rotation: 0, svgOrigin: "540 330" }, { rotation: 30, svgOrigin: "540 330", duration: 1.1, ease: "power2.inOut" }, every + .35);

// Payoff on "minutes": the tram rings in at the top and the town's windows light up.
const done = at("minutes", 4.58) + .2;
tl.to("#tram", { y: 612, duration: .12, yoyo: true, repeat: 1, ease: "power2.out" }, done);
tl.to("#lamp", { fill: "#E7B75F", duration: .08, yoyo: true, repeat: 5, ease: "none" }, done);
tl.to(".house rect:not(:first-child), .house circle", { fill: "#E7B75F", duration: .2, stagger: .02, ease: "power1.out" }, done);
</script>
