<script>
// The streams: widths are the shares said (50 / 30 / 20 of a 300px trunk), each landing on its card.
const X0 = 880, X1 = 1400;
const band = (y0, y1, t0, t1) => `M${X0} ${y0} C ${X0 + 260} ${y0}, ${X1 - 260} ${t0}, ${X1} ${t0} L ${X1} ${t1} C ${X1 - 260} ${t1}, ${X0 + 260} ${y1}, ${X0} ${y1} Z`;
const mid = (y0, y1, t0, t1) => { const a = (y0 + y1) / 2, b = (t0 + t1) / 2; return `M${X0} ${a} C ${X0 + 260} ${a}, ${X1 - 260} ${b}, ${X1} ${b}`; };
const S = [[390, 540, 225, 375], [540, 630, 515, 605], [630, 690, 790, 850]];
S.forEach((v, i) => { document.getElementById('s' + (i + 1)).setAttribute('d', band(...v)); document.getElementById('m' + (i + 1)).setAttribute('d', mid(...v)); });
const NS = 'http://www.w3.org/2000/svg';
// Coins run along each stream once it is open — more on the wide ones.
function runDots(i, t, count, colour) {
  const path = document.getElementById('m' + i), L = path.getTotalLength();
  for (let k = 0; k < count; k++) {
    const c = document.createElementNS(NS, 'circle'); c.setAttribute('r', 7); c.setAttribute('fill', colour); c.setAttribute('opacity', 0);
    document.getElementById('dots').appendChild(c);
    const start = t + k * (1.6 / count);
    for (let s = 0; s <= 10; s++) {
      const p = path.getPointAtLength(L * s / 10);
      if (s === 0) tl.set(c, { attr: { cx: p.x, cy: p.y }, opacity: 0.95 }, start);
      else tl.to(c, { attr: { cx: p.x, cy: p.y }, duration: 0.07, ease: 'none' }, start + (s - 1) * 0.07);
    }
    tl.to(c, { opacity: 0, duration: 0.1 }, start + 0.7);
  }
}
const open = (i, t, card) => {
  tl.to('#r' + i, { attr: { width: 520 }, duration: 0.7, ease: 'power2.inOut' }, t);
  tl.fromTo(card, { opacity: 0, x: 40 }, { opacity: 1, x: 0, duration: 0.45, ease: 'power3.out', immediateRender: false }, t + 0.45);
};
tl.to('#cam', { scale: 1.05, x: -40, duration: DURATION, ease: 'none' }, 0);

// "Every dollar a customer spends" — the customer and the business arrive; a $1 coin arcs from one to the other.
tl.fromTo('#cust', { opacity: 0, y: 30 }, { opacity: 1, y: 0, duration: 0.45, ease: 'power3.out', immediateRender: false }, 0.05);
tl.fromTo('#biz', { opacity: 0, y: 30 }, { opacity: 1, y: 0, duration: 0.45, ease: 'power3.out', immediateRender: false }, 0.25);
const arc = document.getElementById('arc'), AL = arc.getTotalLength();
const tCoin = at('customer', 0.64) - 0.1;
tl.set('#coin', { x: 280, y: 540, opacity: 1, scale: 0.6 }, tCoin);
tl.to('#coin', { scale: 1, duration: 0.2, ease: 'back.out(2)' }, tCoin);
for (let s = 1; s <= 12; s++) { const p = arc.getPointAtLength(AL * s / 12); tl.to('#coin', { x: p.x, y: p.y, duration: 0.06, ease: 'none' }, tCoin + 0.2 + (s - 1) * 0.06); }
tl.to('#coin', { scale: 0.3, opacity: 0, duration: 0.18, ease: 'power2.in' }, tCoin + 0.92);
tl.to('#biz', { scale: 1.06, duration: 0.12, yoyo: true, repeat: 1 }, tCoin + 0.92);

// "goes three ways" — the count sets above the business.
tl.fromTo('#ways', { opacity: 0, y: 16 }, { opacity: 1, y: 0, duration: 0.4, ease: 'back.out(2)', immediateRender: false }, at('3', 1.66) - 0.1);
// "fifty cents to suppliers, thirty to staff, and only twenty…" — each stream opens on its share.
open(1, at('50', 2.52) - 0.3, '#c1'); runDots(1, at('50', 2.52) + 0.3, 6, '#BDEBFF');
open(2, at('30', 4.04) - 0.3, '#c2'); runDots(2, at('30', 4.04) + 0.3, 4, '#D2D8FF');
open(3, at('20', 5.26) - 0.3, '#c3'); runDots(3, at('20', 5.26) + 0.3, 3, '#B6F5D2');
// "…ends up as profit" — everything else steps back; the thin green stream and its card take the light.
const tP = at('ends', 5.58);
tl.to(['#s1', '#s2', '#c1', '#c2'], { opacity: 0.28, duration: 0.4 }, tP);
tl.to('#c3', { scale: 1.08, borderColor: '#2ED67A', boxShadow: '0 20px 40px rgba(0,0,0,.4), 0 0 46px rgba(46,214,122,.35)', duration: 0.4, ease: 'back.out(1.6)' }, tP + 0.1);
tl.to('#c3', { boxShadow: '0 20px 40px rgba(0,0,0,.4), 0 0 80px rgba(46,214,122,.55)', duration: 0.3, yoyo: true, repeat: 1 }, at('profit', 6.36));
</script>
