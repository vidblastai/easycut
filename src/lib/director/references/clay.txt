<script>
// Never still: a slow tabletop push, and the colour behind drifts.
tl.fromTo("#world", { scale: 1.26, transformOrigin: "540px 560px" }, { scale: 1.34, transformOrigin: "540px 560px", duration: DURATION, ease: "none" }, 0);
tl.fromTo("#b1", { x: 0, y: 0 }, { x: 60, y: 40, duration: DURATION, ease: "none" }, 0);
tl.fromTo("#b2", { x: 0 }, { x: -50, duration: DURATION, ease: "none" }, 0);

// Hook: the mound is there; the seed drops in and squashes on "seed".
const seed = Math.min(at("seed", .4), .5) - .3;
tl.fromTo("#mound", { scaleY: .6, svgOrigin: "540 1012" }, { scaleY: 1, duration: .5, ease: "back.out(1.8)" }, 0);
tl.fromTo("#seed", { y: -420, scaleX: .9, scaleY: 1.1, svgOrigin: "540 812" }, { y: 0, duration: .38, ease: "power2.in" }, seed);
tl.to("#seed", { scaleX: 1.18, scaleY: .8, duration: .09, ease: "power1.out" }, seed + .38);
tl.to("#seed", { scaleX: 1, scaleY: 1, duration: .45, ease: "back.out(3)" }, seed + .47);
tl.to("#mound", { scaleY: .94, duration: .09, yoyo: true, repeat: 1, ease: "power1.out" }, seed + .38);

// "water": three drops fall one after another and plop into the seed.
const water = at("water", 1.1) - .25;
["#d0", "#d1", "#d2"].forEach((id, i) => {
  const t = water + i * .16;
  tl.fromTo(id, { y: -260, opacity: 1, scaleY: 1.2, transformOrigin: "50% 100%" }, { y: 440 - (i === 1 ? 0 : 0), duration: .34, ease: "power2.in" }, t);
  tl.to(id, { scaleY: .3, scaleX: 1.6, opacity: 0, duration: .14, ease: "power2.out" }, t + .34);
});
tl.fromTo(".sp", { scale: 0, opacity: 1, transformOrigin: "50% 50%" }, { scale: 1.3, opacity: 0, x: (i) => (i % 2 ? 40 : -40), y: -30, duration: .4, stagger: .05, ease: "power2.out" }, water + .4);
tl.to("#seed", { scaleX: 1.12, scaleY: .9, duration: .1, yoyo: true, repeat: 5, ease: "sine.inOut" }, water + .34);

// "warmth": the clay sun pops in above and glows; the seed swells.
const warm = at("warmth", 1.8) - .15;
tl.fromTo("#sunBody", { scale: 0, svgOrigin: "540 300" }, { scale: 1, duration: .55, ease: "back.out(2)" }, warm);
tl.fromTo("#sunGlow", { scale: 0, opacity: 0, svgOrigin: "540 300" }, { scale: 1, opacity: 1, duration: .7, ease: "expo.out" }, warm + .05);
tl.fromTo(".ray", { scaleY: 0, transformOrigin: "50% 160px" }, { scaleY: 1, duration: .35, stagger: .04, ease: "back.out(2.5)" }, warm + .25);
tl.to("#rays", { rotation: 40, svgOrigin: "540 300", duration: DURATION - warm, ease: "none" }, warm);
tl.to("#seed", { scale: 1.12, svgOrigin: "540 812", duration: .5, ease: "power2.out" }, warm + .2);

// "time": the seed trembles and a crack runs down it.
const time = at("time", 2.36) - .1;
tl.to("#seed", { rotation: 4, duration: .06, yoyo: true, repeat: 7, ease: "sine.inOut", svgOrigin: "540 812" }, time);
tl.set("#crack", { opacity: 1 }, time + .2);
tl.to("#crack", { strokeDashoffset: 0, duration: .4, ease: "power2.inOut" }, time + .2);

// "pushes up": the halves fall open and the stem shoves upward.
const push = at("pushes", 3.3) - .15;
tl.to("#crack", { opacity: 0, duration: .1 }, push);
tl.to("#seedL", { rotation: -32, x: -40, svgOrigin: "540 812", duration: .45, ease: "back.out(2)" }, push);
tl.to("#seedR", { rotation: 32, x: 40, svgOrigin: "540 812", duration: .45, ease: "back.out(2)" }, push);
tl.set("#stem, #stemShine", { opacity: 1 }, push + .05);
tl.to("#stem", { strokeDashoffset: 0, duration: .7, ease: "power3.out" }, push + .05);
tl.to("#stemShine", { strokeDashoffset: 0, duration: .7, ease: "power3.out" }, push + .1);
tl.fromTo("#sprout", { scaleY: .85, svgOrigin: "540 780" }, { scaleY: 1, duration: .5, ease: "back.out(2.4)" }, push + .55);
tl.to("#mound", { scaleY: 1.06, duration: .12, yoyo: true, repeat: 1, ease: "power1.out" }, push);

// "soil" … "light": the leaves unfold and the sprout leans into the sun, which flares.
const leaves = at("soil", 4.3) - .25;
tl.fromTo("#leafL", { scale: 0, rotation: 30, svgOrigin: "546 486" }, { scale: 1, rotation: 0, duration: .55, ease: "back.out(1.8)" }, leaves);
tl.fromTo("#leafR", { scale: 0, rotation: -30, svgOrigin: "552 474" }, { scale: 1, rotation: 0, duration: .55, ease: "back.out(1.8)" }, leaves + .12);
const light = at("light", 5.0) - .2;
tl.to("#sprout", { rotation: -4, svgOrigin: "540 780", duration: .6, ease: "power2.out" }, light);
tl.to("#sunBody", { scale: 1.18, svgOrigin: "540 300", duration: .3, yoyo: true, repeat: 1, ease: "power2.out" }, light);
tl.to("#sunGlow", { scale: 1.35, svgOrigin: "540 300", duration: .5, ease: "expo.out" }, light);
tl.to(".ray", { scaleY: 1.4, duration: .3, stagger: .03, ease: "back.out(2)" }, light);
</script>
