<script>
// Split a line into one span per letter so it can type in; the green part keeps its colour.
function lettersOf(el) {
  const out = [];
  const walk = (node) => {
    [...node.childNodes].forEach((child) => {
      if (child.nodeType === 3) {
        const frag = document.createDocumentFragment();
        [...child.textContent].forEach((c) => { const s = document.createElement('span'); s.className = 'ch'; s.textContent = c; frag.appendChild(s); out.push(s); });
        node.replaceChild(frag, child);
      } else walk(child);
    });
  };
  walk(el);
  return out;
}
function typeIn(el, t, per) {
  const letters = lettersOf(el);
  letters.forEach((s, i) => tl.set(s, { opacity: 1 }, t + i * per));
  return t + letters.length * per;
}
// Something that comes out of the depth: small, blurred and dark to sharp.
function fromDepth(sel, t, d) {
  tl.fromTo(sel, { opacity: 0, scale: 0.55, filter: 'blur(16px)' }, { opacity: 1, scale: 1, filter: 'blur(0px)', duration: d || 0.6, ease: 'power3.out', immediateRender: false }, t);
}
// The camera: a slow constant push, and pans down the canvas as the line moves on.
tl.to('#cam', { scale: 1.05, duration: DURATION, ease: 'none' }, 0);
tl.to('#glow', { scale: 1.08, opacity: 0.9, duration: DURATION / 2, yoyo: true, repeat: 1, ease: 'sine.inOut' }, 0);
const pan = (y, t) => {
  tl.to('#world', { y: -y, duration: 0.95, ease: 'power2.inOut' }, t);
  tl.to('#far', { y: -y * 0.5, duration: 0.95, ease: 'power2.inOut' }, t);
};

// "Three things" — the title types in, with a cursor while it types.
const title = document.getElementById('title');
const cursor = document.createElement('span'); cursor.className = 'cursor';
const end1 = typeIn(title, 0.05, 0.05);
title.appendChild(cursor);
tl.set(cursor, { opacity: 1 }, 0.05);
tl.to(cursor, { opacity: 0, duration: 0.08, repeat: 3, yoyo: true }, end1);
tl.to('#title', { scale: 1.04, duration: 0.5, ease: 'power2.out' }, end1);

// "grew our newsletter" — pan down; the mail tile rises out of the depth.
pan(1080, at('grew', 0.46) - 0.05);
fromDepth('#tile', at('newsletter', 1.08) - 0.25, 0.65);
tl.to('#tilelabel', { opacity: 0.85, duration: 0.3 }, at('newsletter', 1.08) + 0.2);
tl.to('#tile', { boxShadow: '0 0 90px rgba(61,255,110,.55), inset 0 0 40px rgba(61,255,110,.2)', duration: 0.4, yoyo: true, repeat: 1 }, at('50', 1.82));

// "to fifty thousand readers" — the figure types in, then the word under it.
typeIn(document.getElementById('count'), at('50', 1.82) - 0.1, 0.07);
tl.fromTo('#readers', { opacity: 0, y: 20 }, { opacity: 1, y: 0, duration: 0.4, ease: 'power2.out', immediateRender: false }, at('readers', 2.58) - 0.05);

// "one topic, one email a week, and replying to everyone" — pan down; one line per item.
pan(2160, at('readers', 2.58) + 0.35);
const row = (sel, t) => {
  tl.fromTo(sel, { opacity: 0, x: -40, filter: 'blur(10px)' }, { opacity: 1, x: 0, filter: 'blur(0px)', duration: 0.45, ease: 'power3.out', immediateRender: false }, t);
};
row('#r1', at('topic', 3.4) - 0.2);
row('#r2', at('email', 4.26) - 0.15);
row('#r3', at('replying', 5.4) - 0.1);
// The earlier lines settle back as each new one arrives.
tl.to('#r1', { opacity: 0.45, duration: 0.3 }, at('email', 4.26));
tl.to('#r2', { opacity: 0.45, duration: 0.3 }, at('replying', 5.4));
// "everyone" — the last line lights up and a check draws beside it.
const tp = document.getElementById('tickpath');

tl.to('#r3', { color: '#3DFF6E', textShadow: '0 0 30px rgba(61,255,110,.6)', duration: 0.35 }, at('everyone', 6.0));
tl.to(['#r1', '#r2'], { opacity: 0.85, duration: 0.4 }, at('everyone', 6.0) + 0.2);
