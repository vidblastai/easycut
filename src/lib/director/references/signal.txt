// Split a block into one span per word (inside accent spans too), so each word can pop in on its own.
function splitWords(el) {
  const out = [];
  const walk = (node) => {
    [...node.childNodes].forEach((child) => {
      if (child.nodeType !== 3) return walk(child);
      const frag = document.createDocumentFragment();
      child.textContent.split(/(\s+)/).forEach((part) => {
        if (!part) return;
        if (/^\s+$/.test(part)) return frag.appendChild(document.createTextNode(' '));
        const s = document.createElement('span'); s.className = 'w'; s.textContent = part; frag.appendChild(s); out.push(s);
      });
      node.replaceChild(frag, child);
    });
  };
  walk(el);
  return out;
}
// A word pops: up from just below its baseline, out of a blur, with a small settle.
function pop(word, t) {
  tl.fromTo(word, { opacity: 0, y: '0.45em', filter: 'blur(10px)' },
    { opacity: 1, y: 0, filter: 'blur(0px)', duration: 0.5, ease: 'back.out(1.6)', immediateRender: false }, t);
}
// Something that arrives out of the depth: small, blurred and dark to sharp.
function fromDepth(sel, t) {
  tl.fromTo(sel, { opacity: 0, scale: 0.6, filter: 'blur(16px)' }, { opacity: 1, scale: 1, filter: 'blur(0px)', duration: 0.65, ease: 'power3.out', immediateRender: false }, t);
}
// The camera slides sideways to the next window; the world smears horizontally while it moves, and the light drifts the other way.
function slide(x, t, lightX) {
  tl.to('#world', { x: -x, duration: 0.9, ease: 'power3.inOut' }, t);
  tl.to('#far', { x: -x * 0.4, duration: 0.9, ease: 'power3.inOut' }, t);
  tl.to('#mbg', { attr: { stdDeviation: '28 0' }, duration: 0.45, ease: 'power2.in' }, t);
  tl.to('#mbg', { attr: { stdDeviation: '0 0' }, duration: 0.45, ease: 'power2.out' }, t + 0.45);
  tl.to('#light', { x: lightX, duration: 1.4, ease: 'sine.inOut' }, t);
}
// A check draws itself.
function check(row, t) {
  const tick = document.querySelector(row + ' .tick');
  tick.setAttribute('pathLength', '1'); tick.setAttribute('stroke-dasharray', '1'); tick.setAttribute('stroke-dashoffset', '1');
  tl.fromTo(row + ' .check', { opacity: 0, scale: 0.6 }, { opacity: 1, scale: 1, duration: 0.35, ease: 'back.out(1.8)', immediateRender: false }, t);
  tl.to(row + ' .tick', { strokeDashoffset: 0, duration: 0.3, ease: 'power2.out' }, t + 0.12);
}

tl.to('#cam', { scale: 1.05, duration: DURATION, ease: 'none' }, 0);
tl.to('#light', { scale: 1.08, opacity: 0.88, duration: DURATION / 2, yoyo: true, repeat: 1, ease: 'sine.inOut' }, 0);

// "Three things" — each word pops as it is said.
const head = splitWords(document.getElementById('head'));

tl.fromTo('#k', { innerText: 0 }, { innerText: 50, snap: { innerText: 1 }, duration: 0.75, ease: 'power2.out', immediateRender: false }, t50);
tl.to('#tile', { boxShadow: '0 0 90px rgba(70,240,74,.5), inset 0 1px 0 rgba(255,255,255,.12)', duration: 0.35, yoyo: true, repeat: 1 }, t50 + 0.5);
splitWords(document.getElementById('readers')).forEach((w) => pop(w, at('readers', 2.58) - 0.08));

// "one topic, one email a week, and replying to everyone" — slide to the second window; one row per spoken item.
const t2 = at('readers', 2.58) + 0.42;

const rows = [['#r1', at('topic', 3.4) - 0.2], ['#r2', at('email', 4.26) - 0.2], ['#r3', at('replying', 5.4) - 0.12]];

  check(row, t);
  splitWords(document.querySelector(row + ' .txt')).forEach((w, j) => pop(w, t + 0.06 + j * 0.07));
  if (i > 0) tl.to(rows[i - 1][0] + ' .txt', { opacity: 0.5, duration: 0.3 }, t);
