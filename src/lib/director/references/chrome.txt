<script>
// A slow drift, as if orbiting a sculpture; the holographic light breathes.
tl.fromTo("#world", { scale: 1.06, rotation: -1.5 }, { scale: 1.14, rotation: 1.5, transformOrigin: "540px 700px", duration: DURATION, ease: "sine.inOut" }, 0);
tl.fromTo("#holo", { x: -40, rotation: 0 }, { x: 40, rotation: 25, duration: DURATION, ease: "sine.inOut" }, 0);
tl.fromTo("#holo2", { x: 30 }, { x: -60, duration: DURATION, ease: "sine.inOut" }, 0);

// Hook: the drop hangs, wobbling, then falls on "single drop".
tl.fromTo("#drop", { scale: .2, svgOrigin: "540 120" }, { scale: 1, duration: .45, ease: "elastic.out(1, .6)" }, 0);
tl.to("#drop", { scaleX: 1.08, scaleY: .94, svgOrigin: "540 120", duration: .18, yoyo: true, repeat: 3, ease: "sine.inOut" }, .45);
const drop = at("drop", 1.56) - .35;
tl.to("#drop", { y: 260, scaleY: 1.25, scaleX: .85, svgOrigin: "540 120", duration: .32, ease: "power2.in" }, drop);
tl.set("#drop", { opacity: 0 }, drop + .32);
// The pool swells up out of the splash.
tl.set("#pool", { opacity: 1 }, drop + .3);
tl.fromTo("#pool", { scale: .15, scaleY: .05, svgOrigin: "540 700" }, { scale: 1, scaleY: 1, duration: .7, ease: "elastic.out(1, .6)", immediateRender: false }, drop + .3);

// "the beat hits": the pool thumps and throws chrome rings out on the beat.
const beat = at("beat", 2.56) - .05;
tl.set("#rings", { opacity: 1 }, beat);
document.querySelectorAll(".ring").forEach((r, i) => {
  tl.fromTo(r, { scale: .85, opacity: 1, transformOrigin: "540px 700px" }, { scale: 1.25 + i * .1, opacity: 0, duration: .7, ease: "power2.out", immediateRender: false }, beat + i * .12);
});
tl.to("#pool", { scaleX: 1.1, scaleY: .9, svgOrigin: "540 700", duration: .1, yoyo: true, repeat: 3, ease: "power2.out" }, beat);

// "a million people": the pool spits out a cloud of tiny chrome spheres that orbit it.
const many = at("million", 3.38) - .15;
tl.set("#minis", { opacity: 1 }, many);
document.querySelectorAll(".mini").forEach((m, i) => {
  const a = (i / 14) * Math.PI * 2 + rand(i) * .4, d = 360 + rand(i + 9) * 150;
  tl.fromTo(m, { x: 0, y: 0, scale: 0 }, { x: Math.cos(a) * d, y: Math.sin(a) * d * 1.05, scale: 1, duration: .55, ease: "expo.out", immediateRender: false }, many + i * .02);
});
tl.to("#minis", { rotation: 30, svgOrigin: "540 700", duration: DURATION - many, ease: "none" }, many);

// "press play": the pool melts and re-forms as a big inflated chrome play button.
const play = at("press", 3.9) - .05;
tl.to("#pool", { scale: .6, filter: "blur(6px)", svgOrigin: "540 700", duration: .25, ease: "power3.in" }, play);
tl.set("#pool", { opacity: 0 }, play + .25);
tl.set("#play", { opacity: 1 }, play + .2);
tl.fromTo("#play", { scale: .5, filter: "blur(6px)", svgOrigin: "590 650" }, { scale: 1.15, filter: "blur(0px)", duration: .6, ease: "elastic.out(1, .6)", immediateRender: false }, play + .2);
tl.set("#spk", { opacity: 1 }, at("play", 4.24));
tl.fromTo(".sp", { scale: 0, rotation: -45, transformOrigin: "50% 50%" }, { scale: 1, rotation: 0, duration: .3, stagger: .06, ease: "back.out(3)", immediateRender: false }, at("play", 4.24));
tl.to(".sp", { scale: .4, opacity: .6, duration: .2, yoyo: true, repeat: 3, stagger: .05, ease: "sine.inOut" }, at("play", 4.24) + .35);
</script>
