// One span per word (inside the coral span too), so each word can arrive on its own timing.
function splitWords(el) {
  const out = [];
  const walk = (node) => {
    [...node.childNodes].forEach((child) => {
      if (child.nodeType !== 3) return walk(child);
      const frag = document.createDocumentFragment();
      child.textContent.split(/(\s+)/).forEach((part) => {
        if (!part) return;
        if (/^\s+$/.test(part)) return frag.appendChild(document.createTextNode(' '));
        const s = document.createElement('span'); s.className = 'w'; s.textContent = part; frag.appendChild(s); out.push(s);
      });
      node.replaceChild(frag, child);
    });
  };
  walk(el);
  return out;
}
// Calm arrivals: a short rise out of a soft blur, no overshoot.
function rise(sel, t, d, dist) {
  tl.fromTo(sel, { opacity: 0, y: dist || 40, filter: 'blur(14px)' }, { opacity: 1, y: 0, filter: 'blur(0px)', duration: d || 0.6, ease: 'power3.out', immediateRender: false }, t);
}
function leave(sel, t) {
  tl.to(sel, { opacity: 0, y: -30, filter: 'blur(12px)', duration: 0.35, ease: 'power2.in' }, t);
}

tl.to('#cam', { scale: 1.03, duration: DURATION, ease: 'none' }, 0);
tl.to('#lift', { x: 260, y: 80, duration: DURATION, ease: 'sine.inOut' }, 0);

// "Most teams plan the whole year in January." — the quote mark first, then each word as it is said.
rise('#qmark', 0, 0.5, 20);
const said = ['Most', 'teams', 'plan', 'the', 'whole', 'year', 'in', 'January'];
splitWords(document.getElementById('quote')).forEach((w, i) => {
  tl.fromTo(w, { opacity: 0, y: '0.3em', filter: 'blur(6px)' }, { opacity: 1, y: 0, filter: 'blur(0px)', duration: 0.35, ease: 'power3.out', immediateRender: false }, at(said[i], 0.2 * i) - 0.04);
});
const out1 = at('best', 2.56) - 0.3;
leave(['#quote', '#qmark'], out1);

// "The best teams plan…" — the old way as a card; a ✕ badge pops onto its icon on the second "plan" (timed from "best").

tk.setAttribute('pathLength', '1'); tk.setAttribute('stroke-dasharray', '1'); tk.setAttribute('stroke-dashoffset', '1');
tl.fromTo('#badge', { opacity: 0, scale: 0.4 }, { opacity: 1, scale: 1, duration: 0.35, ease: 'back.out(2)', immediateRender: false }, at('time', 4.28) - 0.05);
tl.to('#tick', { strokeDashoffset: 0, duration: 0.3, ease: 'power2.out' }, at('time', 4.28) + 0.08);

// "and get faster, calmer and clearer" — the cards lift away; one pill per spoken word, a dotted connector drawing on to the next.

  if (dots) tl.to(dots, { width: 150, duration: 0.35, ease: 'power2.out' }, t + 0.12);
