<script>
const words = [...document.querySelectorAll('#line .w')];
tl.to('#cam', { scale: 1.04, duration: DURATION, ease: 'none' }, 0);
tl.to('#lift', { opacity: 0.8, scale: 1.05, duration: DURATION / 2, yoyo: true, repeat: 1, ease: 'sine.inOut' }, 0);

// The sentence is there from the first frame, dim; each word lights up as it is said.
tl.fromTo('#line', { opacity: 0, y: 14 }, { opacity: 1, y: 0, duration: 0.35, ease: 'power2.out', immediateRender: false }, 0);
const said = ['Most', 'teams', 'plan', 'the', 'whole', 'year', 'in', 'January'];
said.forEach((w, i) => {
  const t = at(w, 0.2 * i);
  const key = w === 'whole' || w === 'year';
  tl.to(words[i], { color: key ? '#F2775F' : '#F2F2F3', duration: 0.15, ease: 'none' }, t);
});

// "The best teams" — the sentence glitches out: slices offset, then gone.
const out1 = at('best', 2.56) - 0.25;
tl.to('#line', { x: 18, duration: 0.05, yoyo: true, repeat: 3, ease: 'none' }, out1);
tl.to('#line', { opacity: 0, filter: 'blur(8px)', duration: 0.22, ease: 'power2.in' }, out1 + 0.08);

// The old way as a card, struck out; the new way rises beside it.
tl.fromTo('#old', { opacity: 0, y: 24, x: 230 }, { opacity: 1, y: 0, x: 230, duration: 0.4, ease: 'power2.out', immediateRender: false }, out1 + 0.25);
// The second "plan" — at() finds the first, so time it from the word before.
const strike = at('best', 2.56) + 0.42;
tl.set('#stamp', { opacity: 1, x: 230 }, strike);
tl.fromTo('#stamp', { scale: 1.6 }, { scale: 1, duration: 0.18, ease: 'power3.in', immediateRender: false }, strike);
tl.to(['#old', '#stamp'], { x: 0, opacity: 0.55, duration: 0.55, ease: 'power3.inOut' }, at('one', 3.38) - 0.1);
tl.fromTo('#new', { opacity: 0, y: 60 }, { opacity: 1, y: 0, duration: 0.5, ease: 'power3.out', immediateRender: false }, at('week', 3.76) - 0.25);
const nt = document.getElementById('newtick');
nt.setAttribute('pathLength', '1'); nt.setAttribute('stroke-dasharray', '1'); nt.setAttribute('stroke-dashoffset', '1');
tl.to('#newtick', { strokeDashoffset: 0, duration: 0.3, ease: 'power2.out' }, at('time', 4.28));
tl.to('#new', { borderColor: '#3EE07A', boxShadow: '0 30px 60px rgba(0,0,0,.45), 0 0 40px rgba(62,224,122,.25)', duration: 0.3 }, at('time', 4.28));

// "and get faster, calmer and clearer" — the cards dissolve; empty pills appear, then fill as each word is said.
const out2 = at('get', 4.9) - 0.25;
tl.to(['#old', '#stamp', '#new'], { opacity: 0, filter: 'blur(8px)', duration: 0.25, ease: 'power2.in' }, out2);
['#p1', '#p2', '#p3'].forEach((p, i) => tl.fromTo(p, { opacity: 0, y: 16 }, { opacity: 1, y: 0, duration: 0.3, ease: 'power2.out', immediateRender: false }, out2 + 0.2 + i * 0.06));
const fill = (pill, dots, t) => {
  tl.fromTo(pill + ' span', { opacity: 0, y: 10 }, { opacity: 1, y: 0, duration: 0.2, ease: 'power2.out', immediateRender: false }, t);
  tl.to(pill, { borderColor: '#3EE07A', duration: 0.2 }, t);
  if (dots) tl.to(dots, { width: 170, duration: 0.3, ease: 'power2.out' }, t + 0.1);
};
fill('#p1', '#d1', at('faster', 5.06) - 0.05);
fill('#p2', '#d2', at('calmer', 5.68) - 0.05);
fill('#p3', null, at('clearer', 6.16) - 0.05);
tl.to(['#p1', '#p2', '#p3'], { boxShadow: '0 0 36px rgba(62,224,122,.3)', duration: 0.3 }, at('clearer', 6.16) + 0.25);
</script>
